cmake_minimum_required(VERSION 3.20)
set(MYTOOLS_VERSION "0.1.0" CACHE STRING "Version used for builds and release packages")
project(
    mytools
    VERSION ${MYTOOLS_VERSION}
    DESCRIPTION "A small collection of tools that are useful for development purposes"
    HOMEPAGE_URL "https://github.com/kenlooney/mytools"
    LANGUAGES C
)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/bin)

add_subdirectory(tools)

include(GNUInstallDirs)
install(FILES LICENSE README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/mytools)

set(CPACK_PACKAGE_VENDOR "kenlooney")
set(CPACK_PACKAGE_CONTACT "kenlooney")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY ${PROJECT_DESCRIPTION})
set(CPACK_PACKAGE_HOMEPAGE_URL ${PROJECT_HOMEPAGE_URL})
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_SOURCE_DIR}/LICENSE")
set(CPACK_PACKAGE_FILE_NAME "mytools-v${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
set(CPACK_GENERATOR "ZIP")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    list(APPEND CPACK_GENERATOR "DEB" "RPM")
    set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
    set(CPACK_DEBIAN_FILE_NAME "${CPACK_PACKAGE_FILE_NAME}.deb")
    set(CPACK_RPM_PACKAGE_LICENSE "GPL-3.0")
    set(CPACK_RPM_PACKAGE_AUTOREQPROV ON)
    set(CPACK_RPM_FILE_NAME "${CPACK_PACKAGE_FILE_NAME}.rpm")
endif()
include(CPack)
